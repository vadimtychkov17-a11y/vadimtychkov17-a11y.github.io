function updateUI() { document.getElementById('scoreValue').textContent = formatNumber(score); let currentAuto = autoPower * prestigeMultiplier * (1 + talents.auto * 0.5); if (buffTimer > 0) currentAuto *= buffMultiplier; document.getElementById('incomeValue').textContent = formatNumber(currentAuto) + '/сек'; document.getElementById('lvlText').textContent = level; document.getElementById('xpText').textContent = `${Math.floor(xp)} / ${xpNeeded} XP`; document.getElementById('xpFill').style.width = `${Math.min((xp / xpNeeded) * 100, 100)}%`; // Магазин const shopList = document.getElementById('shopList'); if (shopList.children.length === 0) { upgrades.forEach((up, i) => { const div = document.createElement('div'); div.className = 'shop-item'; div.id = `shop-item-${i}`; div.innerHTML = `
${up.icon}
${up.name}
0
0
`; div.onclick = () => buyUpgrade(i); shopList.appendChild(div); }); } upgrades.forEach((up, i) => { const cost = Math.floor(up.baseCost * Math.pow(up.costMult, up.count)); document.getElementById(`cost-${i}`).textContent = formatNumber(cost); document.getElementById(`count-${i}`).textContent = up.count; document.getElementById(`shop-item-${i}`).className = `shop-item ${score >= cost ? '' : 'disabled'}`; }); // Достижения const achList = document.getElementById('achievementsList'); if (achList.children.length === 0) { achievementsList.forEach((ach, i) => { const div = document.createElement('div'); div.className = 'achievement'; div.id = `ach-${i}`; div.innerHTML = `
${ach.name} ${ach.reward > 0 ? `+${formatNumber(ach.reward)}` : ''}
${ach.desc}
0/${ach.target}
`; achList.appendChild(div); }); } achievementsList.forEach((ach, i) => { if (!ach.unlocked) { if (ach.type === 'clicks') ach.progress = totalClicks; if (ach.type === 'score') ach.progress = lifetimeScore; if (ach.type === 'level') ach.progress = level; if (ach.type === 'robot') ach.progress = upgrades.find(u => u.id === 'robot').count; if (ach.type === 'crit') ach.progress = hasCrit ? 1 : 0; if (ach.type === 'golden') ach.progress = hasCaughtGolden ? 1 : 0; document.getElementById(`prog-${i}`).textContent = `${Math.floor(ach.progress)}/${ach.target}`; if (ach.progress >= ach.target) { ach.unlocked = true; if (ach.reward > 0) { score += ach.reward; lifetimeScore += ach.reward; showToast(`🏆 ${ach.name}
Награда: +${formatNumber(ach.reward)}`, '#00ff88'); } else { showToast(`🏆 Достижение: ${ach.name}`, '#00ff88'); } playSound('buy', 1000); } } if (ach.unlocked) document.getElementById(`ach-${i}`).classList.add('unlocked'); }); // ИСПРАВЛЕНИЕ БАГА С МИНУСОВЫМ ПРЕСТИЖЕМ: // Считаем чистую разницу const rawPPoints = Math.floor(Math.sqrt(Math.max(0, lifetimeScore) / 1000000)) - prestigePoints; // Если разница меньше нуля, значит мы просто не заработали новых очков const pPoints = Math.max(0, rawPPoints); const pBtn = document.getElementById('prestigeBtn'); // Показываем кнопку перерождения только если есть смысл (накоплено больше, чем есть) if (pPoints > 0) { pBtn.classList.add('show'); pBtn.textContent = `ПЕРЕРОЖДЕНИЕ (+${pPoints})`; } else { pBtn.classList.remove('show'); } // Дерево талантов теперь зависит от текущего количества очков престижа, а не от будущего перерождения if (prestigePoints > 0) { document.getElementById('talentsContainer').style.display = 'block'; } document.getElementById('talentClick').disabled = prestigePoints < 1; document.getElementById('talentAuto').disabled = prestigePoints < 1; document.getElementById('talentCrit').disabled = prestigePoints < 2; document.getElementById('buffIndicator').style.display = buffTimer > 0 ? 'block' : 'none'; if(buffTimer > 0) document.getElementById('buffIndicator').textContent = `🔥 x${buffMultiplier} ДОХОД! (${buffTimer}с) 🔥`; }
